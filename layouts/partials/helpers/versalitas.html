"H.I.J.O.S." -}} {{- $t = replace $t "Ficha técnica" "Ficha técnica" -}} {{- range (slice "UNSAM" "FADU" "CABA" "CCK" "ACE" "CIC" "UBA" "UNA" "INT") -}} {{- $t = replaceRE (printf `\b%s\b` .) (printf `%s` .) $t -}} {{- end -}} {{- end -}} {{- $t = replaceRE `([^<]+)` `$1` $t -}} {{- $t | safeHTML -}}
