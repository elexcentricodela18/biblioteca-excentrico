{{/* Resuelve URL de video a URL de embed (YouTube, Vimeo). Parámetro: url (string) Salida: solo la URL de embed, o vacío si no se reconoce. */}} {{ $raw := strings.TrimSpace (.url | default "") }} {{ $embed := "" }} {{ if $raw }} {{ if or (strings.Contains $raw "youtube.com/embed/") (strings.Contains $raw "youtube-nocookie.com/embed/") }} {{ $embed = $raw }} {{ else if strings.Contains $raw "player.vimeo.com/video/" }} {{ $embed = $raw }} {{ else if strings.Contains $raw "youtu.be/" }} {{ $id := index (last 1 (split (strings.TrimSuffix "/" $raw) "/")) 0 }} {{ $id = index (split $id "?") 0 }} {{ if $id }}{{ $embed = printf "https://www.youtube.com/embed/%s" $id }}{{ end }} {{ else if strings.Contains $raw "youtube.com/shorts/" }} {{ $id := index (split (index (split $raw "/shorts/") 1) "?") 0 }} {{ if $id }}{{ $embed = printf "https://www.youtube.com/embed/%s" $id }}{{ end }} {{ else if strings.Contains $raw "youtube.com/watch" }} {{ $id := index (split (index (split $raw "v=") 1) "&") 0 }} {{ if $id }}{{ $embed = printf "https://www.youtube.com/embed/%s" $id }}{{ end }} {{ else if strings.Contains $raw "vimeo.com/" }} {{ $id := index (last 1 (split (strings.TrimSuffix "/" $raw) "/")) 0 }} {{ $id = index (split $id "?") 0 }} {{ if and $id (findRE "^[0-9]+$" $id) }}{{ $embed = printf "https://player.vimeo.com/video/%s" $id }}{{ end }} {{ end }} {{ end }} {{ $embed }}
